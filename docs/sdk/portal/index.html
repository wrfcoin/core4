Get Started in Minutes

Choose your platform and start building with WRFCoin SDKs

Java Installation


<dependency>
    <groupId>com.wrfcoin</groupId>
    <artifactId>wrfcoin-sdk</artifactId>
    <version>1.0.0</version>
</dependency>


implementation 'com.wrfcoin:wrfcoin-sdk:1.0.0'

Basic Usage

// Initialize SDK
WRFCoinSDK sdk = new WRFCoinSDK.Builder()
    .apiKey("your-api-key")
    .environment(Environment.PRODUCTION)
    .build();

// Get current weather
CompletableFuture<WeatherData> weather = sdk.getWeatherService()
    .getCurrentWeather(40.7128, -74.0060);

weather.thenAccept(data -> {
    System.out.println("Temperature: " + data.getTemperature() + "°C");
});

C# Installation

# NuGet Package Manager
Install-Package WRFCoin.SDK

# .NET CLI
dotnet add package WRFCoin.SDK

Basic Usage

// Initialize SDK
var sdk = new WRFCoinSDK.Builder()
    .WithApiKey("your-api-key")
    .WithEnvironment(Environment.Production)
    .Build();

// Get current weather
var weather = await sdk.WeatherService
    .GetCurrentWeatherAsync(40.7128, -74.0060);

Console.WriteLine($"Temperature: {weather.Temperature}°C");

Android Installation

// build.gradle.kts (Module: app)
dependencies {
    implementation("com.wrfcoin:sdk-android:1.0.0")
    implementation("org.jetbrains.kotlinx:kotlinx-coroutines-android:1.7.3")
}

Basic Usage

// Initialize in Application class
val config = WRFCoinConfig.Builder()
    .apiKey("your-api-key")
    .environment(WRFCoinEnvironment.PRODUCTION)
    .build()

WRFCoinSDK.initialize(this, config)

// Get current weather
lifecycleScope.launch {
    val weather = WRFCoinSDK.getInstance()
        .weatherService
        .getCurrentWeather(40.7128, -74.0060)
    
    Log.d("Weather", "Temperature: ${weather.temperature}°C")
}

iOS Installation

// Package.swift
dependencies: [
    .package(url: "https://github.com/wrfcoin/ios-sdk", from: "1.0.0")
]

// Or CocoaPods
pod 'WRFCoinSDK', '~> 1.0'

Basic Usage

// Configure in App.swift
let config = WRFCoinConfiguration(
    apiKey: "your-api-key",
    environment: .production
)
WRFCoinSDK.configure(with: config)

// Get current weather
Task {
    do {
        let weather = try await WRFCoinSDK.shared.weatherService
            .getCurrentWeather(latitude: 40.7128, longitude: -74.0060)
        print("Temperature: \(weather.temperature)°C")
    } catch {
        print("Error: \(error)")
    }
}
